\section{Introduction}

\begin{frame}{The Problem}

  \begin{itemize}
    \item Session typed programming languages, such as \textbf{CLASS}, enforce \textbf{memory safety, deadlock-freedom} and other useful properties \textbf{by construction}.
    \item The Linear Session Abstract Machine (LSAM) was introduced to provide a formal model for \textbf{sequentially} executing session-typed programs.
    \item This allowed orders of magnitude faster execution than interpreters which spawn threads for each session.
    \item However, the LSAM remains limited by its high-level nature and the overhead of its interpreter.
  \end{itemize}

\end{frame}

\note{

}

\begin{frame}{Key Idea}

  \begin{itemize}
    \item \textbf{CLASS} is very strongly statically typed, and thus has a lot of information available at compile time.
    \item A compiler could leverage this information to generate efficient native code, while preserving the safety guarantees of the language.
    \item This could enable \textbf{CLASS} and similar languages to be used in performance-critical applications, such as network servers.
  \end{itemize}

\end{frame}

\note{

}

\begin{frame}{Expected Contributions}

  \begin{itemize}
    \item This project introduces a compilation scheme for \textbf{CLASS} programs to native C code.
    \item During this project, a prototype has been implemented which already supports a significant subset of the language.
    \item We aim to extend it to cover the full language, and to use the type information available to implement various optimizations.
    \item Additionally, we will provide a formal proof of correctness for at least the core parts of the compiler.
  \end{itemize}

\end{frame}

\section{Background and Related Work}

\begin{frame}{Background: Linear Logic}
  \begin{itemize}
    \item Introduced by Girard in the 80s \cite{Girard:1987}
    \item \textbf{Propositions are consumed exactly once}, unlike classical logic.
    \item Can be interpreted computationally \cite{Abramsky:1993}, to model concurrent stateful systems.
  \end{itemize}
\end{frame}

\begin{frame}{Background: Linear Logic}
  \begin{itemize}
    \item Connectives: multiplicative ($\otimes$, $\parr$) additive ($\oplus$, $\with$), and exponential ($\oc$, $\wn$).
          \begin{itemize}
            \item Multiplicative connectives model sending and receiving of messages.
            \item Additive connectives model choice and selection.
            \item Exponential connectives model replication and shared state.
          \end{itemize}
    \item Each connective has a dual: $\otimes$ and $\parr$, $\oplus$ and $\with$, $\oc$ and $\wn$.
  \end{itemize}
\end{frame}

\begin{frame}{Background: Process Calculus and Session Types}
  \begin{itemize}
    \item Process calculi are formal languages for describing concurrent systems.
    \item $\pi$-calculus allows mobility of processes \cite{Milner:1992:1}.
          \begin{itemize}
            \item Communication channels can be passed around as values.
            \item Turing complete.
          \end{itemize}
    \item Session types \cite{Honda:1993} restrict communication to predefined protocols.
    \item Operations include sending, receiving, choice, and recursion.
    \item Direct correspondence between session types and linear logic \cite{Caires:2010}.
  \end{itemize}
\end{frame}

\begin{frame}{Background: CLASS}
  \begin{itemize}
    \item CLASS is an experimental language for session-based programs \cite{Rocha:2022}
    \item Expands on the correspondence between session types and linear logic to enable affine types and mutable shared state.
    \item Well-typed programs always terminate, and memory safety is guaranteed.
    \item Allows higher-order polymorphism and even recursive types.
  \end{itemize}
\end{frame}

\note{
  CLASS is a session-typed programming language based on the correspondence between session types and linear logic.

  In fact, CLASS is the first language to simultaneously guarantee memory safety and the absence of deadlocks.
}

\section{Towards a CLASS Compiler}

\begin{frame}{Solution: Overview}
  \begin{itemize}
    \item Compilation scheme heavily based on the SAM interpreter.
    \item Execution is done sequentially and deterministically, with continuation-based coroutining.
    \item Processes communicate in a buffered manner through session records.
    \item CLASS programs are compiled to an Intermediate Representation (IR) which is then translated to C code.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Solution: Session Records}
  \begin{lstlisting}[language=C]

    struct record* records[];
};

struct record {
    void* cont;
    struct environment* cont_env;
    unsigned char cont_record;
    unsigned char written, read;
    unsigned char buffer[];
};

struct environment *env;
  \end{lstlisting}
\end{frame}

\begin{frame}{Solution: Next Steps and Future Work}
  \begin{itemize}
    \item Expand the compiler to cover the full CLASS language.
    \item Deallocate environments using a reference counting scheme.
    \item Profile and optimize the existing implementation.
    \item Explore parallelization using thread pools.
    \item Enable interop with C code.
    \item Emit LLVM IR instead of C code.
  \end{itemize}
\end{frame}
